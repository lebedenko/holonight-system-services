cmake_minimum_required(VERSION 3.25)
project(bluetooth-consumer LANGUAGES CXX)
set(CMAKE_DISABLE_FIND_PACKAGE_PkgConfig TRUE)
find_package(HoloNightSystemServices REQUIRED COMPONENTS Bluetooth)
if(TARGET HoloNightSystem::Audio OR TARGET PkgConfig::LIBPULSE)
  message(FATAL_ERROR "Bluetooth loaded Audio dependencies")
endif()
add_executable(bluetooth-consumer main.cpp)
target_link_libraries(bluetooth-consumer PRIVATE HoloNightSystem::Bluetooth)
